export default function rankStories(stories, builtIn) {
  const ranked = builtIn(stories);
  const isFootball = (story) => story.topics.includes("football");
  const isPremier = (story) => story.competition === "Premier League";

  return [
    ...ranked.filter((story) => isFootball(story) && isPremier(story)),
    ...ranked.filter((story) => isFootball(story) && !isPremier(story)),
    ...ranked.filter((story) => !isFootball(story))
  ];
}
